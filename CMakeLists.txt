cmake_minimum_required(VERSION 3.25)
project(cs2voice VERSION 0.1.5 LANGUAGES CXX)

if(NOT MSVC)
    message(FATAL_ERROR "cs2voice builds with msvc on windows")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)

add_compile_options(/W4 /permissive- /utf-8 /Zc:preprocessor /MP)
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS CS2VOICE_VERSION="${PROJECT_VERSION}"
    CS2VOICE_MAJOR=${PROJECT_VERSION_MAJOR} CS2VOICE_MINOR=${PROJECT_VERSION_MINOR} CS2VOICE_PATCH=${PROJECT_VERSION_PATCH})

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin/$<CONFIG>)

add_subdirectory(core)
add_subdirectory(app)

set_property(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTY VS_STARTUP_PROJECT cs2voice)
